// dot/internal/module/module.go
package module

import (
	"fmt"
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"
)

// Link defines a symlink rule.
type Link struct {
	Source    string   `toml:"source"`
	Target    string   `toml:"target"`
	Platforms []string `toml:"platforms,omitempty"`
}

// PlatformDeps defines package manager dependencies for one platform.
type PlatformDeps struct {
	Brew []string `toml:"brew,omitempty"`
	Apt  []string `toml:"apt,omitempty"`
}

// Deps groups per-platform dependencies.
type Deps struct {
	Darwin PlatformDeps `toml:"darwin"`
	Linux  PlatformDeps `toml:"linux"`
}

// Hooks defines lifecycle shell commands.
type Hooks struct {
	PreInstall  string `toml:"pre_install"`
	PostInstall string `toml:"post_install"`
	PreRemove   string `toml:"pre_remove"`
	PostRemove  string `toml:"post_remove"`
}

// Module represents a parsed module.toml.
type Module struct {
	Name        string   `toml:"name"`
	Description string   `toml:"description"`
	Platforms   []string `toml:"platforms,omitempty"`
	Requires    []string `toml:"requires,omitempty"`
	Exclude     []string `toml:"exclude,omitempty"`
	Links       []Link   `toml:"links"`
	Deps        Deps     `toml:"deps"`
	Hooks       Hooks    `toml:"hooks"`
	Dir         string   `toml:"-"` // filesystem path to module directory
}

// Parse reads and validates a module.toml from the given directory.
func Parse(dir string) (*Module, error) {
	path := filepath.Join(dir, "module.toml")
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("read module.toml: %w", err)
	}

	var mod Module
	if err := toml.Unmarshal(data, &mod); err != nil {
		return nil, fmt.Errorf("parse module.toml: %w", err)
	}

	mod.Dir = dir

	if mod.Name == "" {
		return nil, fmt.Errorf("module.toml: name is required")
	}
	if len(mod.Links) == 0 {
		return nil, fmt.Errorf("module.toml: at least one [[links]] is required")
	}

	return &mod, nil
}
```

package cmd

import (
	"fmt"
	"os"
	"path/filepath"

	"github.com/spf13/cobra"
)

var rootCmd = &cobra.Command{
	Use:   "dot",
	Short: "Dotfiles module manager",
}

// Execute runs the root command.
func Execute() {
	if err := rootCmd.Execute(); err != nil {
		os.Exit(1)
	}
}

// DotfilesPath resolves the dotfiles repository root.
// Looks for the directory containing a "modules/" subdirectory,
// starting from the binary's location, then falling back to ~/dotfiles.
func DotfilesPath() (string, error) {
	// Try binary location first
	exe, err := os.Executable()
	if err == nil {
		dir := filepath.Dir(exe)
		// Walk up to find modules/
		for d := dir; d != "/"; d = filepath.Dir(d) {
			if info, err := os.Stat(filepath.Join(d, "modules")); err == nil && info.IsDir() {
				return d, nil
			}
		}
	}

	// Fallback: ~/dotfiles
	home, err := os.UserHomeDir()
	if err != nil {
		return "", fmt.Errorf("cannot determine home directory: %w", err)
	}
	dfPath := filepath.Join(home, "dotfiles")
	if _, err := os.Stat(filepath.Join(dfPath, "modules")); err != nil {
		return "", fmt.Errorf("dotfiles not found at %s (no modules/ directory)", dfPath)
	}
	return dfPath, nil
}

// dot/internal/module/loader.go
package module

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
)

// LoadAll scans a modules directory and returns all valid modules, sorted by name.
func LoadAll(modulesDir string) ([]*Module, error) {
	entries, err := os.ReadDir(modulesDir)
	if err != nil {
		return nil, fmt.Errorf("read modules directory: %w", err)
	}

	var modules []*Module
	for _, entry := range entries {
		if !entry.IsDir() {
			continue
		}
		dir := filepath.Join(modulesDir, entry.Name())
		if _, err := os.Stat(filepath.Join(dir, "module.toml")); os.IsNotExist(err) {
			continue
		}
		mod, err := Parse(dir)
		if err != nil {
			return nil, fmt.Errorf("module %s: %w", entry.Name(), err)
		}
		modules = append(modules, mod)
	}

	sort.Slice(modules, func(i, j int) bool {
		return modules[i].Name < modules[j].Name
	})

	return modules, nil
}
```

// dot/internal/platform/platform.go
package platform

import "runtime"

// Current returns the current platform identifier: "darwin" or "linux".
func Current() string {
	return runtime.GOOS
}

// MatchesPlatform returns true if the current platform is in the given list.
// An empty or nil list means all platforms.
func MatchesPlatform(platforms []string) bool {
	if len(platforms) == 0 {
		return true
	}
	cur := Current()
	for _, p := range platforms {
		if p == cur {
			return true
		}
	}
	return false
}

// PackageManager returns the package manager command for the current platform.
func PackageManager() string {
	if Current() == "darwin" {
		return "brew"
	}
	return "apt"
}

// dot/internal/manifest/manifest.go
package manifest

import (
	"encoding/json"
	"errors"
	"os"
	"path/filepath"
	"time"
)

// LinkRecord records a single symlink created by a module install.
type LinkRecord struct {
	Source string `json:"source"`
	Target string `json:"target"`
	Backup string `json:"backup,omitempty"` // path to backup file, if any
}

// ModuleRecord records install state for one module.
type ModuleRecord struct {
	InstalledAt string       `json:"installed_at"`
	Links       []LinkRecord `json:"links"`
}

// Manifest tracks all installed modules.
type Manifest struct {
	Version      int                      `json:"version"`
	DotfilesPath string                   `json:"dotfiles_path"`
	Modules      map[string]*ModuleRecord `json:"modules"`
}

// New creates an empty manifest.
func New(dotfilesPath string) *Manifest {
	return &Manifest{
		Version:      1,
		DotfilesPath: dotfilesPath,
		Modules:      make(map[string]*ModuleRecord),
	}
}

// AddModule records a module installation.
func (m *Manifest) AddModule(name string, links []LinkRecord) {
	m.Modules[name] = &ModuleRecord{
		InstalledAt: time.Now().UTC().Format(time.RFC3339),
		Links:       links,
	}
}

// RemoveModule removes a module record.
func (m *Manifest) RemoveModule(name string) {
	delete(m.Modules, name)
}

// IsInstalled checks if a module is recorded as installed.

		return err

	data, err := json.MarshalIndent(m, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(path, data, 0644)
}

// Load reads the manifest from disk. Returns an empty manifest if file doesn't exist.
func Load(path string) (*Manifest, error) {
	data, err := os.ReadFile(path)

		if errors.Is(err, os.ErrNotExist) {
			return New(""), nil
		}
		return nil, err
	}
	var m Manifest
	if err := json.Unmarshal(data, &m); err != nil {
		return nil, err
	}
	if m.Modules == nil {
		m.Modules = make(map[string]*ModuleRecord)
	}
	return &m, nil
}

// dot/internal/linker/linker.go
package linker

import (
	"fmt"
	"os"
	"path/filepath"
	"time"
)

// LinkResult records a created symlink and any backup made.
type LinkResult struct {
	Source     string
	Target     string
	BackupPath string // empty if no backup was made
}

// CreateLink creates a symlink from source to target.
// - Auto-creates parent directories of target.
// - If target exists as a regular file/dir, backs it up.
// - If target exists as a symlink, replaces it.
func CreateLink(source, target string) (*LinkResult, error) {
	// Ensure parent directory
	if err := os.MkdirAll(filepath.Dir(target), 0755); err != nil {
		return nil, fmt.Errorf("mkdir parent: %w", err)
	}

	// Handle existing target
	var backupPath string
	info, err := os.Lstat(target)
	if err == nil {
		if info.Mode()&os.ModeSymlink != 0 {
			// Existing symlink: remove and replace
			os.Remove(target)
		} else {
			// Regular file/dir: backup
			backupPath = fmt.Sprintf("%s.backup.%s", target, time.Now().Format("20060102150405"))
			if err := os.Rename(target, backupPath); err != nil {
				return nil, fmt.Errorf("backup %s: %w", target, err)
			}
		}
	}

	if err := os.Symlink(source, target); err != nil {
		return nil, fmt.Errorf("symlink %s -> %s: %w", target, source, err)
	}

	return &LinkResult{Source: source, Target: target, BackupPath: backupPath}, nil
}

// ExpandDirEntries returns first-level entries in a directory, excluding files matching patterns.
// Uses filepath.Match for pattern matching against file names only.
func ExpandDirEntries(dir string, exclude []string) ([]string, error) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		return nil, err
	}

	var result []string
	for _, entry := range entries {
		if matchesAny(entry.Name(), exclude) {
			continue
		}
		result = append(result, entry.Name())
	}
	return result, nil
}

// matchesAny checks if name matches any of the patterns using filepath.Match.
func matchesAny(name string, patterns []string) bool {
	for _, p := range patterns {
		if matched, _ := filepath.Match(p, name); matched {
			return true
		}
	}
	return false
}

// RemoveLink removes a symlink at the given path.
// Only removes if it is actually a symlink.
func RemoveLink(target string) error {
	info, err := os.Lstat(target)
	if err != nil {
		if os.IsNotExist(err) {
			return nil // already gone
		}
		return err
	}
	if info.Mode()&os.ModeSymlink == 0 {
		return fmt.Errorf("%s is not a symlink, refusing to remove", target)
	}
	return os.Remove(target)
}

// dot/internal/deps/resolver.go
package deps

import "fmt"

// Resolve returns a topologically sorted install order for the requested modules.
// It auto-includes transitive dependencies. Detects cycles.
// allDeps maps module name -> its requires list.
func Resolve(requested []string, allDeps map[string][]string) ([]string, error) {
	visited := map[string]bool{}
	inStack := map[string]bool{}
	var order []string

	var visit func(name string) error
	visit = func(name string) error {
		if inStack[name] {
			return fmt.Errorf("circular dependency detected involving %q", name)
		}
		if visited[name] {
			return nil
		}
		inStack[name] = true
		for _, dep := range allDeps[name] {
			if err := visit(dep); err != nil {
				return err
			}
		}
		inStack[name] = false
		visited[name] = true
		order = append(order, name)
		return nil
	}

	for _, name := range requested {
		if err := visit(name); err != nil {
			return nil, err
		}
	}

	return order, nil
}

// ReverseDeps returns all modules that depend on the given module.
func ReverseDeps(name string, allDeps map[string][]string) []string {
	var result []string
	for mod, deps := range allDeps {
		for _, d := range deps {
			if d == name {
				result = append(result, mod)
				break
			}
		}
	}
	return result
}

	cmd := buildCommand("brew", packages)
	if cmd == nil {
		return nil

	}
	return nil

	cmd := buildCommand("apt-get", packages)
	if cmd == nil {
		return nil
	}
	// apt needs sudo
	args := append([]string{"apt-get", "install", "-y"}, packages...)
	cmd = exec.Command("sudo", args...)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("apt install: %w", err)
	}
	return nil
}

	if len(packages) == 0 {
		return nil
	}
	args := append([]string{"install"}, packages...)
	return exec.Command(pm, args...)
}

	for _, line := range lines {
		line = strings.TrimSpace(line)
		if line != "" {
			files = append(files, line)
		}
	}
	return files, nil
}

// AddAndCommit stages paths and creates a commit.
func AddAndCommit(repoDir string, paths []string, message string) error {
	args := append([]string{"add"}, paths...)
	cmd := exec.Command("git", args...)
	cmd.Dir = repoDir
	if out, err := cmd.CombinedOutput(); err != nil {
		return &GitError{Op: "add", Output: string(out), Err: err}
	}

	cmd = exec.Command("git", "commit", "-m", message)
	cmd.Dir = repoDir
	if out, err := cmd.CombinedOutput(); err != nil {
		return &GitError{Op: "commit", Output: string(out), Err: err}
	}
	return nil
}

			continue
		}

		if mf.IsInstalled(mod.Name) {

			} else {
				fmt.Printf("  %-12s ✓ 已安装 · 无变更\n", mod.Name)
			}
		} else {
			fmt.Printf("  %-12s ✗ 未安装\n", mod.Name)
		}
	}
	return nil
}

// detectChanges checks symlink health and git-tracked file modifications.
func detectChanges(dfPath string, mf *manifest.Manifest, modName string, gitChanges []string) string {
	rec := mf.Modules[modName]
	if rec == nil {
		return ""
	}

	// Check symlink health
	for _, link := range rec.Links {
		info, err := os.Lstat(link.Target)
		if err != nil {
			return "符号链接断裂"
		}
		if info.Mode()&os.ModeSymlink == 0 {
			return "被替换为普通文件"
		}
	}

	// Check git changes in this module's source directory
	prefix := "modules/" + modName + "/"
	var changed []string
	for _, line := range gitChanges {
		parts := strings.Fields(line)
		if len(parts) >= 2 {
			path := parts[len(parts)-1]
			if strings.HasPrefix(path, prefix) {
				fname := filepath.Base(path)
				changed = append(changed, fname)
			}
		}
	}
	if len(changed) > 0 {
		return strings.Join(changed, ", ") + " 已修改"
	}
	return ""
}

package cmd

import (
	"fmt"
	"os"

	"github.com/aporicho/dotfiles/dot/internal/deps"
	gitpkg "github.com/aporicho/dotfiles/dot/internal/git"

	"github.com/aporicho/dotfiles/dot/internal/sysdep"
	"github.com/spf13/cobra"
)

var pullAll bool

var pullCmd = &cobra.Command{
	Use:   "pull [modules...]",
	Short: "Install or update modules",
	RunE:  runPull,
}

func init() {
	pullCmd.Flags().BoolVar(&pullAll, "all", false, "Install/update all modules")
	rootCmd.AddCommand(pullCmd)
}

func runPull(cmd *cobra.Command, args []string) error {
	dfPath, err := DotfilesPath()
	if err != nil {
		return err
	}

	// Pull latest from remote (non-fatal on failure for offline use)
	if err := gitpkg.Pull(dfPath); err != nil {
		fmt.Fprintf(os.Stderr, "警告: git pull 失败: %v（继续使用本地版本）\n", err)
	}

	modulesDir := filepath.Join(dfPath, "modules")
	allModules, err := module.LoadAll(modulesDir)
	if err != nil {
		return err
	}

	// Filter by platform
	var available []*module.Module
	modMap := map[string]*module.Module{}
	depsMap := map[string][]string{}
	for _, mod := range allModules {
		if platform.MatchesPlatform(mod.Platforms) {
			available = append(available, mod)
			modMap[mod.Name] = mod
			depsMap[mod.Name] = mod.Requires
		}
	}

	// Determine which modules to install
	var requested []string
	if pullAll {
		for _, mod := range available {
			requested = append(requested, mod.Name)
		}
	} else if len(args) > 0 {
		requested = args
	} else {
		// No args and no --all: TUI mode (Task 11)
		return fmt.Errorf("请指定模块名或使用 --all 安装全部模块（TUI 模式即将支持）")
	}

	// Validate requested modules exist
	for _, name := range requested {
		if _, ok := modMap[name]; !ok {
			return fmt.Errorf("模块 %q 不存在或不适用于当前平台", name)
		}
	}

	// Resolve dependency order
	order, err := deps.Resolve(requested, depsMap)
	if err != nil {
		return err
	}

	// Load manifest
	mfPath, err := manifest.DefaultPath()
	if err != nil {
		return err
	}
	mf, err := manifest.Load(mfPath)
	if err != nil {
		return err
	}
	mf.DotfilesPath = dfPath

	// Install each module
	for _, name := range order {
		mod := modMap[name]
		if err := installModule(dfPath, mod, mf); err != nil {
			fmt.Fprintf(os.Stderr, "✗ %s: %v\n", name, err)
			continue
		}
		fmt.Printf("✓ %s: 已安装\n", name)
	}

	// Save manifest
	return manifest.Save(mf, mfPath)
}

func installModule(dfPath string, mod *module.Module, mf *manifest.Manifest) error {
	// Pre-install hook
	if err := hook.Run(mod.Hooks.PreInstall, mod.Dir); err != nil {
		return fmt.Errorf("pre_install hook: %w", err)
	}

	// Install system dependencies
	if err := sysdep.Install(mod.Deps.Darwin, mod.Deps.Linux); err != nil {
		fmt.Fprintf(os.Stderr, "  警告: 系统依赖安装失败: %v（继续安装模块）\n", err)
	}

	var records []manifest.LinkRecord
	var results []*linker.LinkResult // for rollback

	rollback := func() {
		for _, r := range results {
			linker.RemoveLink(r.Target)
			// Restore backup if one was made
			if r.BackupPath != "" {
				os.Rename(r.BackupPath, r.Target)
			}
		}
	}

	for _, link := range mod.Links {
		// Platform filter for this link
		if !platform.MatchesPlatform(link.Platforms) {
			continue
		}

		if link.Source == "." {
			// Directory mode: expand entries with exclude
			exclude := mod.Exclude
			if exclude == nil {
				exclude = []string{"module.toml", "README*", "LICENSE*", ".gitignore"}
			}
			entries, err := linker.ExpandDirEntries(mod.Dir, exclude)
			if err != nil {
				rollback()
				return fmt.Errorf("expand dir: %w", err)
			}
			target := expandHome(link.Target)
			for _, entry := range entries {
				src := filepath.Join(mod.Dir, entry)
				tgt := filepath.Join(target, entry)
				result, err := linker.CreateLink(src, tgt)
				if err != nil {
					rollback()
					return err
				}
				results = append(results, result)
				relSrc, _ := filepath.Rel(dfPath, src)
				records = append(records, manifest.LinkRecord{
					Source: relSrc, Target: result.Target, Backup: result.BackupPath,
				})
			}
		} else {
			// Single file/dir mode
			src := filepath.Join(mod.Dir, link.Source)
			tgt := expandHome(link.Target)
			result, err := linker.CreateLink(src, tgt)
			if err != nil {
				rollback()
				return err
			}
			results = append(results, result)
			relSrc, _ := filepath.Rel(dfPath, src)
			records = append(records, manifest.LinkRecord{
				Source: relSrc, Target: result.Target, Backup: result.BackupPath,
			})
		}
	}

	// Post-install hook
	if err := hook.Run(mod.Hooks.PostInstall, mod.Dir); err != nil {
		rollback()
		return fmt.Errorf("post_install hook: %w", err)
	}

	mf.AddModule(mod.Name, records)
	return nil
}

func expandHome(path string) string {
	if strings.HasPrefix(path, "~/") {
		home, _ := os.UserHomeDir()
		return filepath.Join(home, path[2:])
	}
	return path
}

// dot/internal/tui/picker.go
package tui

import tea "github.com/charmbracelet/bubbletea"

// Model is the bubbletea model for the module picker.
type Model struct {
	Items   []ModuleItem
	Cursor  int
	Done    bool
	Quitted bool
}

// Init implements tea.Model.
func (m Model) Init() tea.Cmd { return nil }

// Update implements tea.Model.
func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		switch msg.String() {
		case "q", "ctrl+c", "esc":
			m.Quitted = true
			return m, tea.Quit
		case "up", "k":
			if m.Cursor > 0 {
				m.Cursor--
			}
		case "down", "j":
			if m.Cursor < len(m.Items)-1 {
				m.Cursor++
			}
		case " ":
			m.Items[m.Cursor].Selected = !m.Items[m.Cursor].Selected
		case "enter":
			m.Done = true
			return m, tea.Quit
		case "a":
			m.ToggleAll()
		}
	}
	return m, nil
}

// ToggleAll selects all if any unselected, otherwise deselects all.
func (m *Model) ToggleAll() {
	allSelected := true
	for _, item := range m.Items {
		if !item.Selected {
			allSelected = false
			break
		}
	}
	for i := range m.Items {
		m.Items[i].Selected = !allSelected
	}
}

// SelectedNames returns the names of all selected modules.
func (m Model) SelectedNames() []string {
	var names []string
	for _, item := range m.Items {
		if item.Selected {
			names = append(names, item.Name)
		}
	}
	return names
}

// RunPicker shows a TUI multi-select and returns selected module names.
// Returns nil if user cancelled.
func RunPicker(items []ModuleItem) ([]string, error) {
	m := Model{Items: items}
	p := tea.NewProgram(m)
	result, err := p.Run()
	if err != nil {
		return nil, err
	}
	final := result.(Model)
	if final.Quitted {
		return nil, nil
	}
	return final.SelectedNames(), nil
}

	for _, name := range args {
		if !mf.IsInstalled(name) {
			fmt.Printf("⊘ %s: 未安装，跳过\n", name)
			continue
		}

		// Reverse dependency check
		rdeps := deps.ReverseDeps(name, depsMap)
		var installedRdeps []string
		for _, rd := range rdeps {
			if mf.IsInstalled(rd) {
				installedRdeps = append(installedRdeps, rd)
			}
		}
		if len(installedRdeps) > 0 {
			fmt.Printf("�� %s 被以下模块依赖: %s\n", name, strings.Join(installedRdeps, ", "))
			fmt.Print("确认仍要卸载？[y/N] ")
			reader := bufio.NewReader(os.Stdin)
			answer, _ := reader.ReadString('\n')
			if strings.TrimSpace(strings.ToLower(answer)) != "y" {
				fmt.Println("取消卸载")
				continue
			}
		}

		mod := modMap[name]

		// Pre-remove hook
		if mod != nil {
			if err := hook.Run(mod.Hooks.PreRemove, mod.Dir); err != nil {
				fmt.Fprintf(os.Stderr, "✗ %s: pre_remove hook: %v\n", name, err)
				continue
			}
		}

		// Remove symlinks and restore backups
		rec := mf.Modules[name]
		for _, link := range rec.Links {
			if err := linker.RemoveLink(link.Target); err != nil {
				fmt.Fprintf(os.Stderr, "  警告: 无法移除 %s: %v\n", link.Target, err)
				continue
			}
			// Restore backup if one exists
			if link.Backup != "" {
				if _, err := os.Stat(link.Backup); err == nil {
					if err := os.Rename(link.Backup, link.Target); err != nil {
						fmt.Fprintf(os.Stderr, "  警告: 无法恢复备份 %s: %v\n", link.Backup, err)
					} else {
						fmt.Printf("  ✓ 恢复备份: %s\n", link.Target)
					}
				}
			}
		}

		// Post-remove hook
		if mod != nil {
			if err := hook.Run(mod.Hooks.PostRemove, mod.Dir); err != nil {
				fmt.Fprintf(os.Stderr, "  警告: post_remove hook: %v\n", err)
			}
		}

		mf.RemoveModule(name)
		fmt.Printf("✓ %s: 已卸载\n", name)
	}

	return manifest.Save(mf, mfPath)
}

func runPush(cmd *cobra.Command, args []string) error {
	dfPath, err := DotfilesPath()
	if err != nil {
		return err
	}

	files, err := git.Status(dfPath)
	if err != nil {
		return fmt.Errorf("git status: %w", err)
	}
	if len(files) == 0 {
		fmt.Println("无变更")
		return nil
	}

	fmt.Println("检测到变更：")
	// Group by module
	moduleChanges := map[string][]string{}
	for _, f := range files {
		// Format: "XY path" — extract path part
		parts := strings.Fields(f)
		if len(parts) < 2 {
			continue
		}
		path := parts[len(parts)-1]
		if strings.HasPrefix(path, "modules/") {
			segs := strings.SplitN(path, "/", 3)
			if len(segs) >= 2 {
				modName := segs[1]
				moduleChanges[modName] = append(moduleChanges[modName], path)
			}
		} else {
			moduleChanges["(其他)"] = append(moduleChanges["(其他)"], path)
		}
	}

	for mod, files := range moduleChanges {
		fmt.Printf("  %s:\n", mod)
		for _, f := range files {
			fmt.Printf("    %s\n", f)
		}
	}

	fmt.Print("\n提交并推送？[Y/n] ")
	reader := bufio.NewReader(os.Stdin)
	answer, _ := reader.ReadString('\n')
	answer = strings.TrimSpace(strings.ToLower(answer))
	if answer == "n" || answer == "no" {
		fmt.Println("取消")
		return nil
	}

	// Generate commit message
	msg := pushMessage
	if msg == "" {
		var modNames []string
		for mod := range moduleChanges {
			if mod != "(其他)" {
				modNames = append(modNames, mod)
			}
		}
		if len(modNames) > 0 {
			msg = "update " + strings.Join(modNames, ", ") + " config"
		} else {
			msg = "update dotfiles"
		}
	}

	if err := git.AddAndCommit(dfPath, []string{"modules/"}, msg); err != nil {
		return err
	}
	fmt.Printf("✓ 已提交: %s\n", msg)

	if err := git.Push(dfPath); err != nil {
		return fmt.Errorf("推送失败: %w", err)
	}
	fmt.Println("✓ 已推送到远端")
	return nil
}

package cmd

import (

	Short: "Add local config to dotfiles management",

func runAdd(cmd *cobra.Command, args []string) error {
	dfPath, err := DotfilesPath()
	if err != nil {
		return err
	}

	srcPath, err := filepath.Abs(args[0])
	if err != nil {
		return err
	}

	info, err := os.Stat(srcPath)
	if err != nil {
		return fmt.Errorf("路径不存在: %s", srcPath)
	}

	// Determine module name
	modName := addName
	if modName == "" {
		if info.IsDir() {
			modName = filepath.Base(srcPath)
		} else {
			return fmt.Errorf("单文件需要通过 --name 指定模块名")
		}
	}

	modDir := filepath.Join(dfPath, "modules", modName)
	if _, err := os.Stat(modDir); err == nil {
		return fmt.Errorf("模块 %s 已存在，如需更新请直接编辑", modName)
	}

	// List what we'll add
	if info.IsDir() {
		entries, _ := os.ReadDir(srcPath)
		fmt.Printf("检测到 %s 配置：\n", modName)
		for _, e := range entries {
			eInfo, _ := e.Info()
			if eInfo != nil {
				fmt.Printf("  %s (%s)\n", e.Name(), formatSize(eInfo.Size()))
			}
		}
	} else {
		fmt.Printf("检测到文件: %s\n", filepath.Base(srcPath))
	}

	fmt.Printf("\n创建模块 modules/%s？[Y/n] ", modName)
	reader := bufio.NewReader(os.Stdin)
	answer, _ := reader.ReadString('\n')
	if strings.TrimSpace(strings.ToLower(answer)) == "n" {
		fmt.Println("取消")
		return nil
	}

	// Copy files
	if err := os.MkdirAll(modDir, 0755); err != nil {
		return err
	}

	if info.IsDir() {
		if err := copyDir(srcPath, modDir); err != nil {
			return fmt.Errorf("复制目录: %w", err)
		}
	} else {
		data, err := os.ReadFile(srcPath)
		if err != nil {
			return err
		}
		if err := os.WriteFile(filepath.Join(modDir, filepath.Base(srcPath)), data, info.Mode()); err != nil {
			return err
		}
	}

	// Generate module.toml
	targetPath := srcPath
	home, _ := os.UserHomeDir()
	if strings.HasPrefix(targetPath, home) {
		targetPath = "~" + targetPath[len(home):]
	}

	mod := module.Module{
		Name:        modName,
		Description: modName + " configuration",
	}

	if info.IsDir() {
		mod.Links = []module.Link{{Source: ".", Target: targetPath}}
		mod.Exclude = []string{"module.toml", "README*"}
	} else {
		mod.Links = []module.Link{{Source: filepath.Base(srcPath), Target: targetPath}}
	}

	f, err := os.Create(filepath.Join(modDir, "module.toml"))
	if err != nil {
		return err
	}
	defer f.Close()
	if err := toml.NewEncoder(f).Encode(mod); err != nil {
		return err
	}

	fmt.Printf("✓ 文件已复制到 modules/%s/\n", modName)
	fmt.Println("✓ module.toml 已生成")

	// Replace original with symlink
	parsedMod, err := module.Parse(modDir)
	if err != nil {
		return err
	}

	mfPath, _ := manifest.DefaultPath()
	mf, _ := manifest.Load(mfPath)
	mf.DotfilesPath = dfPath

	if err := installModule(dfPath, parsedMod, mf); err != nil {
		return fmt.Errorf("创建符号链接: %w", err)
	}
	if err := manifest.Save(mf, mfPath); err != nil {
		return err
	}

	fmt.Println("✓ 符号链接已创建")
	return nil
}

func copyDir(src, dst string) error {
	entries, err := os.ReadDir(src)
	if err != nil {
		return err
	}
	for _, entry := range entries {
		srcPath := filepath.Join(src, entry.Name())
		dstPath := filepath.Join(dst, entry.Name())
		if entry.IsDir() {
			os.MkdirAll(dstPath, 0755)
			if err := copyDir(srcPath, dstPath); err != nil {
				return err
			}
		} else {
			data, err := os.ReadFile(srcPath)
			if err != nil {
				return err
			}
			info, _ := entry.Info()
			if err := os.WriteFile(dstPath, data, info.Mode()); err != nil {
				return err
			}
		}
	}
	return nil
}

func formatSize(bytes int64) string {
	if bytes < 1024 {
		return fmt.Sprintf("%dB", bytes)
	}
	return fmt.Sprintf("%.1fKB", float64(bytes)/1024)
}

// dot/cmd/doctor.go
package cmd

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/aporicho/dotfiles/dot/internal/manifest"
	"github.com/spf13/cobra"
)

var doctorCmd = &cobra.Command{
	Use:   "doctor",
	Short: "Check symlink health for installed modules",
	RunE:  runDoctor,
}

func init() {
	rootCmd.AddCommand(doctorCmd)
}

func runDoctor(cmd *cobra.Command, args []string) error {
	dfPath, err := DotfilesPath()
	if err != nil {
		return err
	}

	mfPath, err := manifest.DefaultPath()
	if err != nil {
		return err
	}
	mf, err := manifest.Load(mfPath)
	if err != nil {
		return err
	}

	if len(mf.Modules) == 0 {
		fmt.Println("没有已安装的模块")
		return nil
	}

	fmt.Println("检查模块健康状态...")
	issues := 0
	var broken []string

	for name, rec := range mf.Modules {
		modIssues := []string{}

		for _, link := range rec.Links {
			info, err := os.Lstat(link.Target)
			if err != nil {
				modIssues = append(modIssues, fmt.Sprintf("符号链接缺失: %s", link.Target))
				continue
			}
			if info.Mode()&os.ModeSymlink == 0 {
				modIssues = append(modIssues, fmt.Sprintf("被替换为普通文件: %s", link.Target))
				continue
			}
			// Check symlink points to correct source
			resolved, err := os.Readlink(link.Target)
			if err != nil {
				modIssues = append(modIssues, fmt.Sprintf("无法读取符号链接: %s", link.Target))
				continue
			}
			expectedSrc := filepath.Join(dfPath, link.Source)
			if resolved != expectedSrc {
				modIssues = append(modIssues, fmt.Sprintf("指向错误: %s -> %s (应为 %s)", link.Target, resolved, expectedSrc))
			}
		}

		if len(modIssues) > 0 {
			fmt.Printf("  %-12s ✗ %s\n", name, modIssues[0])
			for _, issue := range modIssues[1:] {
				fmt.Printf("  %-12s   %s\n", "", issue)
			}
			issues += len(modIssues)
			broken = append(broken, name)
		} else {
			fmt.Printf("  %-12s ✓ 正常\n", name)
		}
	}

	if issues > 0 {
		fmt.Printf("\n发现 %d 个问题。运行 dot pull %s 修复。\n", issues, strings.Join(broken, " "))
	} else {
		fmt.Println("\n所有模块状态正常。")
	}
	return nil
}
